% A signed cell of S.  Sign is carried by the printed sign, not by fill.
\newcommand{\cell}[3]{%
  \ifnum#3=0
    \node[draw=dGray!55,fill=white,line width=0.3pt,minimum size=4.2mm,
      inner sep=0pt,font=\dSmall,text=dGray] at ({#1},{#2}) {$\cdot$};
  \else
    \ifnum#3>0
      \node[draw=dGray!55,fill=white,line width=0.3pt,minimum size=4.2mm,
        inner sep=0pt,font=\dSmall,text=dInk] at ({#1},{#2}) {$+#3$};
    \else
      \node[draw=dGray!55,fill=white,line width=0.3pt,minimum size=4.2mm,
        inner sep=0pt,font=\dSmall,text=dInk] at ({#1},{#2}) {$#3$};
    \fi
  \fi}
